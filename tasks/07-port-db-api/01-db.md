```csharp
// What the library needs from a context. DbContext already has every member:
// a project writes `class MyContext : DbContext, IDbContext` and adds nothing.
public interface IDbContext
{
    DbSet<T> Set<T>() where T : class;

    EntityEntry<T> Entry<T>(T entity) where T : class;

    ChangeTracker ChangeTracker { get; }

    DatabaseFacade Database { get; }

    Task<int> SaveChangesAsync(CancellationToken cancellationToken = default);
}

// src/Zerox.Core/Data/IRepository.cs — namespace Zerox.Core.Data (no EF type in it: IQueryable, Task)
public interface IRepository<T> where T : class, IEntity
{
    // Not tracked: a read has no reason to keep rows watched. `tracked: true` only to change a
    // collection of children from the parent (removed rows are deleted only if the parent is tracked).
    IQueryable<T> Query(bool tracked = false);

    // What this unit of work holds and is NOT on the database yet (the same row arriving twice in one
    // sync page must find the first one).
    IEnumerable<T> Local();

    Task<T?> GetByIdAsync(int id, CancellationToken ct = default);   // tracked

    Task AddAsync(T entity, CancellationToken ct = default);

    Task AddRangeAsync(IEnumerable<T> entities, CancellationToken ct = default);

    void Update(T entity);

    void Delete(T entity);

    void DeleteRange(IEnumerable<T> entities);

    // Writes — or is swallowed when a boundary is open (block 02). The caller does not know which.
    Task<int> SaveChangesAsync(CancellationToken ct = default);
}

public interface IRepositoryFactory
{
    // For a base that is not generic with a generic method; whoever can inject IRepository<T> does.
    IRepository<T> For<T>() where T : class, IEntity;
}
```

```csharp
namespace Zerox.Db;

// The state of ONE context instance. It lives and dies with the context (ConditionalWeakTable).
internal sealed class DbSession
{
    private static readonly ConditionalWeakTable<object, DbSession> Sessions = new();

    internal static DbSession Of(object context) => Sessions.GetValue(context, static _ => new DbSession());
}

internal sealed class SessionInterceptor : SaveChangesInterceptor
{
    internal static readonly SessionInterceptor Instance = new();

    private SessionInterceptor() { }

    public override InterceptionResult<int> SavingChanges(DbContextEventData eventData, InterceptionResult<int> result) =>
        throw new NotSupportedException(
            "Save only with SaveChangesAsync: the synchronous door would bypass the unit of work.");

    public override ValueTask<InterceptionResult<int>> SavingChangesAsync(
        DbContextEventData eventData, InterceptionResult<int> result, CancellationToken cancellationToken = default) =>
        ValueTask.FromResult(result);   // blocks 02-04 add the rules
}

public static class DbContextOptionsBuilderExtensions
{
    public static DbContextOptionsBuilder UseZeroxSession(this DbContextOptionsBuilder builder)
    {
        ArgumentNullException.ThrowIfNull(builder);
        return builder.AddInterceptors(SessionInterceptor.Instance);
    }

    internal static bool HasZeroxSession(IDbContext context) =>
        ((IInfrastructure<IServiceProvider>)context).Instance.GetService<IDbContextOptions>()?
            .FindExtension<CoreOptionsExtension>()?.Interceptors?.Contains(SessionInterceptor.Instance) == true;
}
```

```csharp
namespace Zerox.Db;

internal sealed class Repository<T> : IRepository<T> where T : class, IEntity
{
    private readonly IDbContext _db;

    public Repository(IDbContext db)
    {
        ArgumentNullException.ThrowIfNull(db);
        if (!DbContextOptionsBuilderExtensions.HasZeroxSession(db))
            throw new InvalidOperationException(
                "The context is not configured with UseZeroxSession(): the unit of work would not swallow the saves.");
        _db = db;
    }

    public IQueryable<T> Query(bool tracked = false) => tracked ? _db.Set<T>() : _db.Set<T>().AsNoTracking();
    public IEnumerable<T> Local() => _db.Set<T>().Local;
    public async Task<T?> GetByIdAsync(int id, CancellationToken ct = default) =>
        await _db.Set<T>().FindAsync([id], ct).ConfigureAwait(false);
    public async Task AddAsync(T entity, CancellationToken ct = default) =>
        await _db.Set<T>().AddAsync(entity, ct).ConfigureAwait(false);
    public Task AddRangeAsync(IEnumerable<T> entities, CancellationToken ct = default) => _db.Set<T>().AddRangeAsync(entities, ct);
    public void Update(T entity) => _db.Set<T>().Update(entity);
    public void Delete(T entity) => _db.Set<T>().Remove(entity);
    public void DeleteRange(IEnumerable<T> entities) => _db.Set<T>().RemoveRange(entities);
    public Task<int> SaveChangesAsync(CancellationToken ct = default) => _db.SaveChangesAsync(ct);
}

internal sealed class RepositoryFactory(IDbContext db) : IRepositoryFactory
{
    public IRepository<T> For<T>() where T : class, IEntity => new Repository<T>(db);
}

public static class ServiceCollectionExtensions
{
    public static IServiceCollection AddZeroxDb<TContext>(
        this IServiceCollection services, Action<DbContextOptionsBuilder> configure)
        where TContext : DbContext, IDbContext
    {
        ArgumentNullException.ThrowIfNull(services);
        ArgumentNullException.ThrowIfNull(configure);
        services.AddDbContext<TContext>(options =>
        {
            configure(options);
            options.UseZeroxSession();
        });
        services.AddScoped<IDbContext>(sp => sp.GetRequiredService<TContext>());
        services.AddScoped(typeof(IRepository<>), typeof(Repository<>));
        services.AddScoped<IRepositoryFactory, RepositoryFactory>();
        return services;
    }
}
```

## TDDAB-2: Add the Boundary (IUnitOfWork) That Swallows the Saves

<intro>
Delivers the contract `IUnitOfWork` and `IRepository<T>.OpenUnitOfWork()` (in `src/Zerox.Core/Data`), and
their implementation on block 01's `DbSession` and `SessionInterceptor` (`src/Zerox.Db`).

Semantics (from Filo, unchanged):
- `OpenUnitOfWork()` opens the boundary on the context: the first one is the ROOT; one opened while a
  root is open is a SLAVE (its `ApplyAsync` does nothing and returns 0; its dispose does nothing);
- inside an open boundary every `SaveChangesAsync` (through a repository OR directly on the context)
  returns 0 and writes nothing: the interceptor returns `InterceptionResult<int>.SuppressWithResult(0)`;
- `ApplyAsync` on the root: closes the boundary, then ONE `SaveChangesAsync` with all the work;
  applying the same root twice → `InvalidOperationException`;
- if the applying save throws, `ChangeTracker.Clear()` and rethrow: the work of a dead boundary must
  never reach the database later, piecemeal, through somebody else's save;
- dispose without applying: the boundary closes and nothing is written (nothing to roll back: nothing
  reached the database). After apply or dispose, saves write again.
- It is NOT a transaction: no connection is held while the logic runs.

`DbSession` gains `UnitOfWork? Boundary` and `bool Applying`; `SessionInterceptor.SavingChangesAsync`
gains the swallow rule. `UnitOfWork` is `internal sealed`, a thin handle that forwards to the session
(the context's own `DisposeAsync` is taken, so the handle must be another object).

`<filo>/src/Filo.Infrastructure/Data/FiloDbContext.cs` (`ApriConfine`, `ApplicaConfineAsync`,

```csharp
// src/Zerox.Core/Data/IUnitOfWork.cs — namespace Zerox.Core.Data
// The boundary of an operation that writes more than one repository: inside, saves are swallowed;
// ApplyAsync writes once, with all the work. Not a transaction: nothing reaches the database before it.
public interface IUnitOfWork : IAsyncDisposable
{
    // On a slave (opened inside another boundary) does nothing: the root writes.
    Task<int> ApplyAsync(CancellationToken ct = default);
}

// IRepository<T> gains:
//   IUnitOfWork OpenUnitOfWork();
// Repository<T>: public IUnitOfWork OpenUnitOfWork() => DbSession.Of(_db).OpenBoundary(_db);
```

```csharp
internal sealed class DbSession
{
    // ... Of(...) from block 01
    internal UnitOfWork? Boundary { get; private set; }
    internal bool Applying { get; private set; }

    internal IUnitOfWork OpenBoundary(IDbContext db)
    {
        if (Boundary is not null) return new UnitOfWork(this, db, slave: true);
        Boundary = new UnitOfWork(this, db, slave: false);
        return Boundary;
    }

    internal async Task<int> ApplyAsync(UnitOfWork who, IDbContext db, CancellationToken ct)
    {
        if (who.Slave) return 0;
        if (!ReferenceEquals(who, Boundary))
            throw new InvalidOperationException("This unit of work has already been applied or closed.");

        Boundary = null;
        Applying = true;
        try
        {
            return await db.SaveChangesAsync(ct).ConfigureAwait(false);
        }
        catch
        {
            // The boundary is dead and its work is still in the tracker: without this, the next save
            // that passes would write it piecemeal — half an operation on the database.
            db.ChangeTracker.Clear();
            throw;
        }
        finally
        {
            Applying = false;
        }
    }

    internal void CloseWithoutApplying(UnitOfWork who)
    {
        if (ReferenceEquals(who, Boundary)) Boundary = null;
    }
}

// SessionInterceptor.SavingChangesAsync:
//   var session = DbSession.Of(eventData.Context!);
//   if (session.Boundary is not null && !session.Applying)
//       return ValueTask.FromResult(InterceptionResult<int>.SuppressWithResult(0));
//   return ValueTask.FromResult(result);
```

Because the boundary is cleared before the applying save, `Applying` is only needed to keep the
rule readable; the swallow condition is `Boundary is not null`. Keep both: the test
`Should_BeASlave_WhenOpenedInsideAnotherBoundary` must stay green if the order inside `ApplyAsync`
changes.

`UnitOfWork(DbSession session, IDbContext db, bool slave)`: `ApplyAsync` → `session.ApplyAsync(this, db, ct)`
and marks itself closed; `DisposeAsync` → if not closed, `session.CloseWithoutApplying(this)`;
returns `ValueTask.CompletedTask`.

## TDDAB-3: Add the Exclusive Massive Operation

<intro>
Delivers the contracts `IMassiveOpHandler<T>`, `IFieldsToSet<T>` and `IRepository<T>.OpenMassiveAsync()`
(in `src/Zerox.Core/Data`) and the implementation in `src/Zerox.Db`, on block 01's session and block 02's
boundary.

Semantics (from Filo, unchanged): update or delete many rows WITHOUT loading them
(`ExecuteUpdateAsync`/`ExecuteDeleteAsync`), which reach the database at once and bypass the tracker.
So the massive operation is EXCLUSIVE:
- it does not open with pending changes (`InvalidOperationException`), nor inside an open boundary
  (`InvalidOperationException`), nor while another one is open (`InvalidOperationException`);
- opening it clears the tracker (entities read before would be stale) and begins a transaction —
  the ONLY transaction of the library, used only by it;
- while it is open, every save throws `InvalidOperationException` (rule in `SessionInterceptor`);
- `CommitAsync` commits and closes; dispose without commit rolls back and closes.
`IFieldsToSet<T>.Set(field, value)` declares constant values; `MassiveOpHandler<T>` translates them to
EF's `UpdateSettersBuilder<T>` (the contract stays free of EF builder types).

`DbSession` gains `object? Massive` and `IDbContextTransaction? MassiveTransaction`.

```csharp
// src/Zerox.Core/Data/IMassiveOpHandler.cs, IFieldsToSet.cs — namespace Zerox.Core.Data
// Updates or deletes many rows without loading them. Exclusive: see the block intro.
public interface IMassiveOpHandler<T> : IAsyncDisposable where T : class
{
    Task<int> UpdateWhereAsync(Expression<Func<T, bool>> where, Action<IFieldsToSet<T>> fields, CancellationToken ct = default);

    Task<int> DeleteWhereAsync(Expression<Func<T, bool>> where, CancellationToken ct = default);

    // Without it, dispose rolls back.
    Task CommitAsync(CancellationToken ct = default);
}

// The fields a massive update sets, declared without the writer's types (keeps the contract free of
// EF's builder). One method because there is one case: a constant value.
public interface IFieldsToSet<T>
{
    IFieldsToSet<T> Set<TField>(Expression<Func<T, TField>> field, TField value);
}

// IRepository<T> gains:
//   Task<IMassiveOpHandler<T>> OpenMassiveAsync(CancellationToken ct = default);
```

```csharp
// DbSession
internal async Task<IMassiveOpHandler<T>> OpenMassiveAsync<T>(IDbContext db, CancellationToken ct) where T : class
{
    if (Massive is not null) throw new InvalidOperationException("A massive operation is already open.");
    if (Boundary is not null) throw new InvalidOperationException("A unit of work is open: massive writes do not mix with tracked ones.");
    if (db.ChangeTracker.HasChanges()) throw new InvalidOperationException("There are pending changes: save them before a massive operation.");

    db.ChangeTracker.Clear();
    MassiveTransaction = await db.Database.BeginTransactionAsync(ct).ConfigureAwait(false);
    var handler = new MassiveOpHandler<T>(this, db);
    Massive = handler;
    return handler;
}
// CommitMassiveAsync(who, ct): ReferenceEquals check → commit → CloseMassiveAsync(who)
// CloseMassiveAsync(who): ReferenceEquals check (else return) → dispose transaction (rolls back if not
// committed) → Massive = null

// SessionInterceptor.SavingChangesAsync, FIRST rule after the sync one:
//   if (session.Massive is not null) throw new InvalidOperationException(
//       "A massive operation is open: no repository can save until it is closed.");
```

`MassiveOpHandler<T>`: `UpdateWhereAsync` builds a `FieldsToSet<T>` (private nested class collecting
`Action<UpdateSettersBuilder<T>>` steps, each `b => b.SetProperty(field, value)`), then
`db.Set<T>().Where(where).ExecuteUpdateAsync(fields.Apply, ct)`; `DeleteWhereAsync` →
`ExecuteDeleteAsync`; `CommitAsync` → `session.CommitMassiveAsync(this, ct)`; `DisposeAsync` →
`session.CloseMassiveAsync(this)`.

```csharp
// Opt-in optimistic lock: Version is the concurrency token; configuration (ApplyOptimisticLock) and
// increment (the session interceptor) are done for you. On conflict EF throws
// DbUpdateConcurrencyException; LastWriterWins (Zerox.Domain) re-reads and retries.
public interface IHasOptimisticLock
{
    int Version { get; set; }
}

// src/Zerox.Db/ModelBuilderExtensions.cs — namespace Zerox.Db
public static class ModelBuilderExtensions
{
    public static ModelBuilder ApplyOptimisticLock(this ModelBuilder modelBuilder)
    {
        ArgumentNullException.ThrowIfNull(modelBuilder);
        foreach (var type in modelBuilder.Model.GetEntityTypes()
                     .Where(t => typeof(IHasOptimisticLock).IsAssignableFrom(t.ClrType)))
            modelBuilder.Entity(type.ClrType).Property(nameof(IHasOptimisticLock.Version)).IsConcurrencyToken();
        return modelBuilder;
    }
}
```
